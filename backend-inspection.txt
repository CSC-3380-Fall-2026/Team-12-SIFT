import os

import psycopg
from dotenv import load_dotenv
from pydantic import BaseModel, Field


load_dotenv()


class AnnotationCreate(BaseModel):
    user_id: str
    media_id: str
    text: str = Field(min_length=1)
    x: float
    y: float


class Annotation(AnnotationCreate):
    id: int


def get_db_connection():
    return psycopg.connect(
        host="localhost",
        port=5432,
        dbname="sift",
        user="postgres",
        password=os.getenv("POSTGRES_PASSWORD"),
        connect_timeout=5,
    )


def create_annotation(annotation: AnnotationCreate) -> Annotation:
    with get_db_connection() as connection:
        with connection.cursor() as cursor:
            cursor.execute(
                """
                INSERT INTO annotations (user_id, media_id, text, x, y)
                VALUES (%s, %s, %s, %s, %s)
                RETURNING id
                """,
                (
                    annotation.user_id,
                    annotation.media_id,
                    annotation.text,
                    annotation.x,
                    annotation.y,
                ),
            )

            annotation_id = cursor.fetchone()[0]

    return Annotation(id=annotation_id, **annotation.model_dump())

def get_annotations(media_id: str) -> list[Annotation]:
    with get_db_connection() as connection:
        with connection.cursor() as cursor:
            cursor.execute(
                """
                SELECT id, user_id, media_id, text, x, y
                FROM annotations
                WHERE media_id = %s
                ORDER BY id
                """,
                (media_id,),
            )

            rows = cursor.fetchall()

    return [
        Annotation(
            id=row[0],
            user_id=row[1],
            media_id=row[2],
            text=row[3],
            x=row[4],
            y=row[5],
        )
        for row in rows
    ]
class AnnotationUpdate(BaseModel):
    text: str = Field(min_length=1)
    x: float
    y: float


def update_annotation(
    annotation_id: int,
    update: AnnotationUpdate,
) -> Annotation | None:
    with get_db_connection() as connection:
        with connection.cursor() as cursor:
            cursor.execute(
                """
                UPDATE annotations
                SET text = %s, x = %s, y = %s
                WHERE id = %s
                RETURNING id, user_id, media_id, text, x, y
                """,
                (
                    update.text,
                    update.x,
                    update.y,
                    annotation_id,
                ),
            )

            row = cursor.fetchone()

    if row is None:
        return None

    return Annotation(
        id=row[0],
        user_id=row[1],
        media_id=row[2],
        text=row[3],
        x=row[4],
        y=row[5],
    )
def delete_annotation(annotation_id: int) -> bool:
    with get_db_connection() as connection:
        with connection.cursor() as cursor:
            cursor.execute(
                """
                DELETE FROM annotations
                WHERE id = %s
                RETURNING id
                """,
                (annotation_id,),
            )

            deleted_row = cursor.fetchone()

    return deleted_row is not None

from fastapi import FastAPI, HTTPException

from backend.app.annotations import (
    Annotation,
    AnnotationCreate,
    AnnotationUpdate,
    create_annotation,
    get_annotations,
    update_annotation,
    delete_annotation,
)


app = FastAPI()

media_storage: dict[str, list[str]] = {}

@app.get("/")
def root():
    return {"message": "Backend for SIFT project is running"}


@app.post("/annotations", response_model=Annotation, status_code=201)
def create_annotation_endpoint(annotation: AnnotationCreate):
    return create_annotation(annotation)


@app.get("/annotations/{media_id}", response_model=list[Annotation])
def get_annotations_endpoint(media_id: str):
    return get_annotations(media_id)


@app.put("/annotations/{annotation_id}", response_model=Annotation)
def update_annotation_endpoint(
    annotation_id: int,
    update: AnnotationUpdate,
):
    annotation = update_annotation(annotation_id, update)

    if annotation is None:
        raise HTTPException(status_code=404, detail="Annotation not found")

    return annotation


@app.delete("/annotations/{annotation_id}")
def delete_annotation_endpoint(annotation_id: int):
    deleted = delete_annotation(annotation_id)

    if not deleted:
        raise HTTPException(status_code=404, detail="Annotation not found")

    return {"message": "Annotation deleted successfully"}
